cmake_minimum_required(VERSION 3.20)
project(pnp-readiness-v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
function(warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
add_library(pnp-core-v2 pnp_coverage_v2.cpp hardware_ids_v2.cpp)
warnings(pnp-core-v2)
add_executable(pnp-policy-v2 pnp_coverage_v2_tests.cpp)
target_link_libraries(pnp-policy-v2 PRIVATE pnp-core-v2)
warnings(pnp-policy-v2)
add_test(NAME pnp-policy-v2 COMMAND pnp-policy-v2)
add_test(NAME pnp-evidence-v2 COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/pnp_coverage_v2_tests.py $<TARGET_FILE:pnp-policy-v2>)
set_tests_properties(pnp-policy-v2 pnp-evidence-v2 PROPERTIES TIMEOUT 45)
if(WIN32)
    add_library(pnp-windows-v2 pnp_windows_v2.cpp)
    target_compile_definitions(pnp-windows-v2 PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(pnp-windows-v2 PUBLIC pnp-core-v2 setupapi cfgmgr32)
    warnings(pnp-windows-v2)
    target_link_libraries(pnp-policy-v2 PRIVATE pnp-windows-v2)
    add_executable(pnp-coverage-v2 pnp_observer_v2.cpp)
    target_link_libraries(pnp-coverage-v2 PRIVATE pnp-windows-v2)
    warnings(pnp-coverage-v2)
    if(MINGW)
        target_link_options(pnp-coverage-v2 PRIVATE -municode -static)
    endif()
endif()

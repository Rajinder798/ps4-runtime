cmake_minimum_required(VERSION 3.24)
project(pnp-rejection-v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_subdirectory(../no_guest_coordinator_v1 coordinator)
add_library(pnp-rejection-core-v1 rejection_v1.cpp)
target_link_libraries(pnp-rejection-core-v1 PUBLIC pnp-core-v2)
warnings(pnp-rejection-core-v1)
add_executable(pnp-rejection-host-v1 rejection_host_v1.cpp)
target_link_libraries(pnp-rejection-host-v1 PRIVATE pnp-rejection-core-v1)
warnings(pnp-rejection-host-v1)
if(WIN32)
    add_library(pnp-rejection-windows-v1 rejection_windows_v1.cpp)
    target_compile_definitions(pnp-rejection-windows-v1 PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(pnp-rejection-windows-v1 PUBLIC pnp-rejection-core-v1 pnp-windows-v2)
    warnings(pnp-rejection-windows-v1)
    add_executable(pnp-rejection-windows-link-v1 rejection_windows_link_v1.cpp)
    target_link_libraries(pnp-rejection-windows-link-v1 PRIVATE pnp-rejection-windows-v1)
    warnings(pnp-rejection-windows-link-v1)
    add_test(NAME pnp-rejection-windows-link-v1 COMMAND pnp-rejection-windows-link-v1)
endif()
add_test(NAME pnp-rejection-v1 COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/rejection_tests.py $<TARGET_FILE:pnp-rejection-host-v1>)
set_tests_properties(pnp-rejection-v1 PROPERTIES TIMEOUT 60)

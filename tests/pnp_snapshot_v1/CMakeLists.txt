cmake_minimum_required(VERSION 3.24)
project(pnp-snapshot-v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    OUTPUT_VARIABLE SNAPSHOT_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
configure_file(snapshot_build_v1.h.in snapshot_build_v1.h @ONLY)
add_subdirectory(../pnp_rejection_v1 rejection)
add_library(pnp-snapshot-core-v1 snapshot_v1.cpp)
target_include_directories(pnp-snapshot-core-v1 PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pnp-snapshot-core-v1 PUBLIC pnp-rejection-core-v1)
warnings(pnp-snapshot-core-v1)
add_executable(pnp-snapshot-host-v1 snapshot_host_v1.cpp)
target_link_libraries(pnp-snapshot-host-v1 PRIVATE pnp-snapshot-core-v1)
warnings(pnp-snapshot-host-v1)
add_executable(pnp-snapshot-budget-v1 snapshot_budget_tests.cpp)
target_link_libraries(pnp-snapshot-budget-v1 PRIVATE pnp-snapshot-core-v1)
warnings(pnp-snapshot-budget-v1)
add_test(NAME snapshot-budget-v1 COMMAND pnp-snapshot-budget-v1)
if(WIN32)
    add_executable(pnp-snapshot-windows-v1 snapshot_windows_v1.cpp)
    target_link_libraries(pnp-snapshot-windows-v1 PRIVATE pnp-snapshot-core-v1 pnp-rejection-windows-v1)
    warnings(pnp-snapshot-windows-v1)
    if(MINGW)
        target_link_options(pnp-snapshot-windows-v1 PRIVATE -municode -static)
    endif()
    add_test(NAME snapshot-inert-link-v1 COMMAND pnp-snapshot-windows-v1 --inert-link-check)
    set(inert $<TARGET_FILE:coordinator-inert-v1>)
else()
    set(inert none)
endif()
add_test(NAME snapshot-contract-v1 COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/snapshot_tests.py $<TARGET_FILE:pnp-snapshot-host-v1> ${inert})
set_tests_properties(snapshot-contract-v1 PROPERTIES TIMEOUT 90)

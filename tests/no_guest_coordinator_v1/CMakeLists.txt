cmake_minimum_required(VERSION 3.24)
project(no-guest-coordinator-v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_subdirectory(../pnp_readiness_v2 pnp-v2)
if(WIN32)
    add_executable(endpoint-clock-v1 endpoint_clock_v1.cpp)
    target_link_libraries(endpoint-clock-v1 PRIVATE ole32 uuid)
    add_executable(coordinator-inert-v1 WIN32 coordinator_inert_v1.cpp)
    target_link_libraries(coordinator-inert-v1 PRIVATE shell32)
    foreach(target endpoint-clock-v1 coordinator-inert-v1)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
            target_link_options(${target} PRIVATE -municode -static)
        endif()
    endforeach()
    set(guard $<TARGET_FILE:endpoint-clock-v1>)
    set(inert $<TARGET_FILE:coordinator-inert-v1>)
else()
    set(guard none)
    set(inert none)
endif()
add_test(NAME no-guest-protocol-v1 COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/coordinator_tests.py $<TARGET_FILE:pnp-policy-v2>)
add_test(NAME no-guest-ownership-v1 COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/coordinator_windows_tests.py ${inert} ${guard})
set_tests_properties(no-guest-protocol-v1 no-guest-ownership-v1 PROPERTIES TIMEOUT 60)
